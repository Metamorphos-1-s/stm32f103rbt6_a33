# Host-only counterexamples. These policies are never firmware sources.
get_target_property(research_sources stage5p_unified_smooth_tests SOURCES)
get_target_property(research_includes stage5p_unified_smooth_tests INCLUDE_DIRECTORIES)
get_target_property(research_defs stage5p_unified_smooth_tests COMPILE_DEFINITIONS)
get_target_property(research_source_dir stage5p_unified_smooth_tests SOURCE_DIR)
set(research_common)
foreach(source IN LISTS research_sources)
    if(source MATCHES "test_unified.c$|test_defaults_battery.c$|a13c_shadow_compensator.c$")
        continue()
    endif()
    get_filename_component(full_source "${source}" ABSOLUTE BASE_DIR "${research_source_dir}")
    list(APPEND research_common "${full_source}")
endforeach()
get_filename_component(controller "${CMAKE_CURRENT_SOURCE_DIR}/../../../Domain/measurement/a13c_shadow_compensator.c" ABSOLUTE)
file(READ "${controller}" controller_text)
foreach(policy baseline hold60 hold120 paired_reference)
    set(policy_source "${controller}")
    set(extra_source)
    if(policy STREQUAL hold60 OR policy STREQUAL hold120)
        if(policy STREQUAL hold60)
            set(hold 600U)
        else()
            set(hold 1200U)
        endif()
        string(REPLACE "#define HOLD_SAMPLES 150U" "#define HOLD_SAMPLES ${hold}"
            generated_text "${controller_text}")
        if(generated_text STREQUAL controller_text)
            message(FATAL_ERROR "Frozen hold definition changed")
        endif()
        set(policy_source "${CMAKE_CURRENT_BINARY_DIR}/${policy}_controller.c")
        file(WRITE "${policy_source}" "${generated_text}")
    elseif(policy STREQUAL paired_reference)
        string(REPLACE "bool A13C_Feed(" "bool R5ResearchOriginalFeed("
            generated_text "${controller_text}")
        if(generated_text STREQUAL controller_text)
            message(FATAL_ERROR "Frozen controller feed signature changed")
        endif()
        set(policy_source "${CMAKE_CURRENT_BINARY_DIR}/${policy}_controller.c")
        file(WRITE "${policy_source}" "${generated_text}")
        set(extra_source paired_reference_policy.c)
    endif()
    set(target stage5p_r5_transition_${policy})
    add_executable(${target} runner.c ${research_common} ${policy_source} ${extra_source})
    target_include_directories(${target} PRIVATE ${research_includes})
    target_compile_definitions(${target} PRIVATE ${research_defs}
        A33_ENABLE_STAGE5P_SAVE_CONTINUITY=1)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
